import java.util.Scanner;

public class IT23172746Lab5Q3 {
    public static void main(String[] args) {
        // Constants
        final double ROOM_CHARGE = 48000.00;
        final double DISCOUNT_10 = 0.10;
        final double DISCOUNT_20 = 0.20;
        
        Scanner input = new Scanner(System.in);
        
        System.out.print("Enter Start Date (1-31): ");
        int startDate = input.nextInt();
        
        System.out.print("Enter End Date (1-31): ");
        int endDate = input.nextInt();
        
        // Validation 1: Dates must be between 1 and 31
        if (startDate < 1 || startDate > 31 || endDate < 1 || endDate > 31) {
            System.out.println("Error: Days must be between 1 and 31");
            return;
        }
        
        // Validation 2: Start date must be before end date
        if (startDate >= endDate) {
            System.out.println("Error: Start Date must be less than End Date");
            return;
        }
        
        // Calculate days reserved
        int days = endDate - startDate;
        
        // Determine discount rate
        double discountRate = 0;
        if (days >= 5) {
            discountRate = DISCOUNT_20;
        } else if (days >= 3) {
            discountRate = DISCOUNT_10;
        }
        
        // Calculate total
        double totalAmount = days * ROOM_CHARGE * (1 - discountRate);
        
        // Print results
        System.out.println();
        System.out.println("Room Charge Per Day: Rs. " + ROOM_CHARGE + "/=");
        System.out.println("Number of Days Reserved: " + days);
        System.out.println("Total Amount to be Paid: " + totalAmount);
    }
}
